# Comandos básicos do GIT
git clone https://github.com/aluno/repositorio.git
git remote remove origin
git remote add origin https://github.com/aluno/repositorio.git
git push -u origin main


git init ou git init nome-do-seu-projeto
git config --global user.email "seu-email@exemplo.com"
git config --global user.name "seu-nome-usuario"
git status
git add . ou git add nome-do-seu-arquivo-com-a-extensão
git commit -m "sua mensagem para referenciar o que subiu no servidor"
git push

git pull
